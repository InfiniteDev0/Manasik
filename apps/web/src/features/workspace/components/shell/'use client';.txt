'use client';

import { useEffect, useState } from 'react';
import { useParams, useRouter, useSearchParams } from 'next/navigation';
import confetti from 'canvas-confetti';
import { toast, Toaster } from 'sonner';
import { WelcomeDialog } from '@/components/workspace/WelcomeDialog';
import { useAuthStore } from '@/lib/store';
import ClientNavbar from '../../../components/workspace/clientnav';
import LiquidGlassNav from '@/components/workspace/liquidglassnav';
import WorkspaceSidebar from '@/components/workspace/workspace-sidebar';
import BottomPill from '@/features/workspace/components/shell/BottomPill';

export default function WorkspaceLayout({ children }: { children: React.ReactNode }) {
  const params = useParams();
  const router = useRouter();
  const searchParams = useSearchParams();

  const workspaceId = params.workspaceId as string;
  const isWelcome = searchParams.get('welcome') === '1';

  const [showWelcome, setShowWelcome] = useState(false);
  const [welcomeDismissed, setWelcomeDismissed] = useState(false);

  useEffect(() => {
    if (!isWelcome) {
      return;
    }

    // Strip ?welcome=1 from the URL immediately
    router.replace(`/workspace/${workspaceId}`, { scroll: false });

    // Side-cannon confetti after 300ms
    const cannon = setTimeout(() => {
      const end = Date.now() + 3000;
      const colors = ['#22c55e', '#ffffff', '#a3e635', '#86efac'];
      const frame = () => {
        if (Date.now() > end) {
          return;
        }
        confetti({
          particleCount: 2,
          angle: 60,
          spread: 55,
          startVelocity: 60,
          origin: { x: 0, y: 0.5 },
          colors,
        });
        confetti({
          particleCount: 2,
          angle: 120,
          spread: 55,
          startVelocity: 60,
          origin: { x: 1, y: 0.5 },
          colors,
        });
        requestAnimationFrame(frame);
      };
      frame();
    }, 300);

    // Toast after 500ms
    const toastTimer = setTimeout(() => {
      const user = useAuthStore.getState().user;
      toast.message(
        `Welcome to Lenzro${user?.name ? `, ${user.name}` : ''}! Your workspace is ready.`,
        { duration: 5000, position: 'bottom-left' }
      );
    }, 500);

    // Welcome dialog after 800ms
    const dialog = setTimeout(() => setShowWelcome(true), 800);

    return () => {
      clearTimeout(cannon);
      clearTimeout(toastTimer);
      clearTimeout(dialog);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isWelcome]);

  return (
    <div className="min-h-screen bg-background">
      <div className="bg-background h-screen overflow-hidden flex flex-col">
        {/* Sticky top bar */}
        <div className="sticky top-0 z-40">
          <ClientNavbar />
        </div>

        {/* Sidebar + page content */}
        <div className="flex flex-row flex-1 gap-1 p-1 min-h-0">
          <WorkspaceSidebar />
          <div className="flex-1 scrollbar-pill custom-scrollbar min-h-0 rounded-md border workspace overflow-auto">
            {children}
          </div>
        </div>
      </div>

      {/* Floating AI assistant */}
      <div className="fixed bottom-8 left-1/2 -translate-x-1/2 z-50">
        <BottomPill/>
      </div>

      <div className="fixed bottom-8 right-3 z-50"></div>

      <Toaster position="bottom-left" />

      <WelcomeDialog
        open={showWelcome && !welcomeDismissed}
        workspaceName={workspaceId}
        workspaceSlug={workspaceId}
        plan="FREE"
        toolsCount={5}
        onClose={() => {
          setWelcomeDismissed(true);
          setShowWelcome(false);
        }}
      />
    </div>
  );
}
